void BasicC::Read_Buttons(){
    int SW[6]={0,0,0,0,0,0};

    Wire.beginTransmission(0x20);       //address first port expander
    Wire.write(0x00);                   // select input register
    Wire.endTransmission(false);        // send repeated start instead of stop
    Wire.requestFrom(0x20, 1);          // Start a read access and expect one byte
    int db_Switches = Wire.read();              // read inputs
    Wire.endTransmission();             //now send a stop

    for (int i = 2; i < 8; i++) {
        byte Switches_new = db_Switches ;
        byte Swi_byte = Switches_new >> i;
        int Swi_byte_new = Swi_byte;
        if(i>4){
            if (Swi_byte_new % 2 == 0) {SW[i-2] = 0 ;}
            else                       {SW[i-2] = 1 ;}
        }
        else{
            if (Swi_byte_new % 2 == 0) {SW[i-2] = 1 ;}
            else                       {SW[i-2] = 0 ;}
        }
    }

    Switch1 = SW[5];
    Switch2 = SW[4];
    Switch3 = SW[3];

    Button1 = SW[0];
    Button2 = SW[1];
    Button3 = SW[2];
}

void BasicC::Read_US(){
    int reading;
    if(US_Timer>100){

        // US Sensoren auslesen
        Wire.beginTransmission(113); // transmit to device #112
        Wire.write(byte(0x02));      // sets register pointer to echo #1 register (0x02)
        Wire.endTransmission();      // stop transmitting
        Wire.requestFrom(113, 2);    // request 2 bytes from slave device #112
        if (2 <= Wire.available()) { // if two bytes were received
        reading = Wire.read();  // receive high byte (overwrites previous reading)
        reading = reading << 8;    // shift high byte to be high 8 bits
        reading |= Wire.read(); // receive low byte as lower 8 bits
        US_F = reading;
        }
        Wire.beginTransmission(114); // transmit to device #112
        Wire.write(byte(0x02));      // sets register pointer to echo #1 register (0x02)
        Wire.endTransmission();      // stop transmitting
        Wire.requestFrom(114, 2);    // request 2 bytes from slave device #112
        if (2 <= Wire.available()) { // if two bytes were received
        reading = Wire.read();  // receive high byte (overwrites previous reading)
        reading = reading << 8;    // shift high byte to be high 8 bits
        reading |= Wire.read(); // receive low byte as lower 8 bits
        US_R=reading;
        }
        Wire.beginTransmission(115); // transmit to device #112
        Wire.write(byte(0x02));      // sets register pointer to echo #1 register (0x02)
        Wire.endTransmission();      // stop transmitting
        Wire.requestFrom(115, 2);    // request 2 bytes from slave device #112
        if (2 <= Wire.available()) { // if two bytes were received
        reading = Wire.read();  // receive high byte (overwrites previous reading)
        reading = reading << 8;    // shift high byte to be high 8 bits
        reading |= Wire.read(); // receive low byte as lower 8 bits
        US_L=reading;
        }
        Wire.beginTransmission(116); // transmit to device #112
        Wire.write(byte(0x02));      // sets register pointer to echo #1 register (0x02)
        Wire.endTransmission();      // stop transmitting
        Wire.requestFrom(116, 2);    // request 2 bytes from slave device #112
        if (2 <= Wire.available()) { // if two bytes were received
        reading = Wire.read();  // receive high byte (overwrites previous reading)
        reading = reading << 8;    // shift high byte to be high 8 bits
        reading |= Wire.read(); // receive low byte as lower 8 bits
        US_B=reading;
        }

        // US Sensoren Anfragen
        Wire.beginTransmission(113); // transmit to device #112 (0x70)
        Wire.write(byte(0x00));      // sets register pointer to the command register (0x00)
        Wire.write(byte(0x50));      // command sensor to measure in "inches" (0x50)
        Wire.endTransmission();      // stop transmitting
        Wire.beginTransmission(114); // transmit to device #112 (0x70)
        Wire.write(byte(0x00));      // sets register pointer to the command register (0x00)
        Wire.write(byte(0x50));      // command sensor to measure in "inches" (0x50)
        Wire.endTransmission();      // stop transmitting
        Wire.beginTransmission(115); // transmit to device #112 (0x70)
        Wire.write(byte(0x00));      // sets register pointer to the command register (0x00)
        Wire.write(byte(0x50));      // command sensor to measure in "inches" (0x50)
        Wire.endTransmission();      // stop transmitting
        Wire.beginTransmission(116); // transmit to device #112 (0x70)
        Wire.write(byte(0x00));      // sets register pointer to the command register (0x00)
        Wire.write(byte(0x50));      // command sensor to measure in "inches" (0x50)
        Wire.endTransmission();      // stop transmitting

        US_Timer = 0;
    }

    Position = 0;
}

void BasicC::Read_Compas(){ 
    uint8_t system2 = 0;
    uint8_t gyro = 0;
    uint8_t accel = 0;
    uint8_t mag = 0;
    bno.getCalibration(&system2, &gyro, &accel, &mag);
    
    sensors_event_t event; 
    bno.getEvent(&event);
    int Comp_Dir = round(event.orientation.x);

    Comp_Dir = Comp_Dir-Comp_Offset;
    if (Comp_Dir<0){Comp_Dir = Comp_Dir+360;}
    else if (Comp_Dir>359){Comp_Dir = Comp_Dir-360;}

    if(Comp_Dir<=180){Comp_Dir=Comp_Dir;}
    else if (Comp_Dir>180){Comp_Dir=Comp_Dir-360;}
    if (Comp_Dir<-180){Comp_Dir = Comp_Dir+360;}
    else if (Comp_Dir>359){Comp_Dir = Comp_Dir-360;}

    Orbit = Comp_Dir;
    Cal_Status = mag;
}

void BasicC::Calculate_PID(){
    float Setpoint = PID_IN; 

    float Input = Orbit;

    float e = Input-Setpoint; 

    esum = e + esum;

    if (esum > 205){esum =205;}
    else if (esum < -205){esum =-205;}

    if (e > -(12) && e < 12){esum = 0;}

    float out = ((Kp) * e ) + ((Ki)  * esum * 0.45) + ((Kd) * (e - ealt)) ;
    ealt = e;

    PID_OUT = out;
}

void BasicC::Calculate_Ball_Position(){
    Ball_x = 0;
    Ball_y = 0;
    
    for (int i = 0; i < IR_Number; i++) {
        Ball_x = Ball_x + IR_Val[i]*cosf(Gewicht[i]*PI/180);
        Ball_y = Ball_y + IR_Val[i]*sinf(Gewicht[i]*PI/180);
    }

    Ball_Heading = (atan2f(Ball_y,Ball_x+10)*180/PI); 

    Ball_Relative = -(Ball_Heading-Orbit);

    int Ball_Relative2 = Ball_Relative+375;

    if (Ball_Relative2 > 360){
        Ball_Relative2 = Ball_Relative2-360;
    }

    Ball_Raw = round(map(Ball_Relative2,0,360,0,12)); 
    
    for (int i = 0; i < 8; i++)
    {
        if (IR_Val[i] > maxValue)
        {
            secondmaxValue = maxValue;
            maxValue = IR_Val[i];
        }
        else if (IR_Val[i] > secondmaxValue)
        {
            secondmaxValue = IR_Val[i];
        }
    }
    
    DistanceSum = maxValue + secondmaxValue;
    Distance = map(DistanceSum, 0, Ball_Calibrate, 0, 100);

    Ball_Dist_Percentage = Distance+sin(((Ball_Heading+11.25)*8)*PI/180)*40;

    if (Ball_Heading < 0){
        BallDirpercentage = map(-Ball_Heading, 0, 180, 0, 80) + 20; 
    }
    else{
        BallDirpercentage = map(Ball_Heading, 0, 180, 0, 80) + 20; 
    }
}

void BasicC::Calculate_Ball_Anfahrt(){
    DistanceCalcFaktor = 1 + (Distance * BallDirpercentage * DistanceCalcBase);
    Ball_Anfahrt = DistanceCalcFaktor * Ball_Relative;
}

void BasicC::Action_Motor(int motor,bool Dir,int Speed){
    Wire.beginTransmission(0x21); //address second port expander
    Wire.write(0x01);             // select output register
    Wire.endTransmission(false);  // send restart instead of stop
    Wire.requestFrom(0x21, 1);    // Start a read access and expect one byte
    //Wire.read();                  // get current port state
    Wire.endTransmission();
    
    if(((motor==0)&&(Dir==true))||(motor0_U == 1)){
        motor0_N = B00000010;
        motor0_U = 1;
    }
    if (((motor==0)&&(Dir==false))||(motor0_U == 2)){
        motor0_N = B00000001;
        motor0_U = 2;
    }

    if (((motor==1)&&(Dir==true))||(motor1_U == 1)){
        motor1_N = B00000100;
        motor1_U = 1;
    }
    if (((motor==1)&&(Dir==false))||(motor1_U == 2)){
        motor1_N = B00001000;
        motor1_U = 2;
    }

    if (((motor==2)&&(Dir==true))||(motor2_U == 1)){
        motor2_N = B00010000;
        motor2_U = 1;
    }
    if (((motor==2)&&(Dir==false))||(motor2_U == 2)){
        motor2_N = B00100000;
        motor2_U = 2;
    }

    if (((motor==3)&&(Dir==true))||(motor3_U == 1)){
        motor3_N = B01000000;
        motor3_U = 1;
    }
    if (((motor==3)&&(Dir==false))||(motor3_U == 2)){
        motor3_N = B10000000;
        motor3_U = 2;
    }

    if(motor==0){analogWrite(4,Speed);}
    if(motor==1){analogWrite(5,Speed);}
    if(motor==2){analogWrite(6,Speed);}
    if(motor==3){analogWrite(9,Speed);}

    byte OUT = motor0_N | motor1_N | motor2_N | motor3_N  ;
    Wire.beginTransmission(0x21); //address second port expander
    Wire.write(0x01);             // select output register
    Wire.write(OUT);            // update output ports Port_Seting
    Wire.endTransmission();
}

void BasicC::Action_Move(int Dir,int Speed,int Turn){
    PID_IN = Turn;

    Dir = Dir-Turn;

    Speed = Speed*2.55;

    float TurnFormal =  PID_OUT * PID_Multiplier;
    
    float rad = Dir*PI/180;
    float CoorX = sinf(rad);
    float CoorY = cosf(rad);

    if (Speed == 0){
    Velocity_D[0] = (TurnFormal*PID_Stand_Multiplier);
    Velocity_D[1] = (TurnFormal*PID_Stand_Multiplier);
    Velocity_D[2] = (TurnFormal*PID_Stand_Multiplier);
    }
    else{
    Velocity_D[0] = ((((-0.5*CoorX)+(sqrt3/2 * CoorY))) + TurnFormal);
    Velocity_D[1] = ((((-0.5*CoorX)-(sqrt3/2 * CoorY))) + TurnFormal);
    Velocity_D[2] = ((CoorX                           ) + TurnFormal);
    }

    int Dir0;
    int Dir1;
    int Dir2;
    float Error;

    // Alle werte auf + setzen
    if (Velocity_D[0]<0){Velocity[0]=Velocity_D[0]*-1;Dir0=0;}
    else{Velocity[0]=Velocity_D[0];Dir0=1;}
    if (Velocity_D[1]<0){Velocity[1]=Velocity_D[1]*-1;Dir1=0;}
    else{Velocity[1]=Velocity_D[1];Dir1=1;}
    if (Velocity_D[2]<0){Velocity[2]=Velocity_D[2]*-1;Dir2=0;}
    else{Velocity[2]=Velocity_D[2];Dir2=1;}

    // Werte Mappen auf "Speed"
    if (Speed != 0){
    if ((Velocity[0]>=Velocity[1]) && (Velocity[0]>=Velocity[2])){
        float Error = Speed/Velocity[0];
        Velocity[0]= Velocity[0]*Error;
        Velocity[1]= Velocity[1]*Error;
        Velocity[2]= Velocity[2]*Error;
    }
    if ((Velocity[1]>=Velocity[0]) && (Velocity[1]>=Velocity[2])){
        Error = Speed/Velocity[1];
        Velocity[0]= Velocity[0]*Error;
        Velocity[1]= Velocity[1]*Error;
        Velocity[2]= Velocity[2]*Error;
    }
    if ((Velocity[2]>=Velocity[1]) && (Velocity[2]>=Velocity[0])){
        Error = Speed/Velocity[2];
        Velocity[0]= Velocity[0]*Error;
        Velocity[1]= Velocity[1]*Error;
        Velocity[2]= Velocity[2]*Error;
    }
    }

    if (Dir0==1){Action_Motor(0,false,Velocity[0]);}
    else {Action_Motor(0,true,(Velocity[0]));}
    if (Dir1==1){Action_Motor(1,false,Velocity[1]);}
    else {Action_Motor(1,true,(Velocity[1]));}
    if (Dir2==1){Action_Motor(2,false,Velocity[2]);}
    else {Action_Motor(2,true,(Velocity[2]));}
}

void BasicC::Schuss(){
    if (SchussTimer < 500)
    {
        Wire.beginTransmission(0x20); // address second port expander
        Wire.write(0x01);             // select output register
        Wire.write(0xFD);             // Bit 0 on
        Wire.endTransmission();
    }
    else {
        Wire.beginTransmission(0x20); // address second port expander
        Wire.write(0x01);             // select output register
        Wire.write(0x00);            // Bit 0 on
        Wire.endTransmission();
    }
    
}
