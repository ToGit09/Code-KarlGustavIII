// Handelt das Display
class CodeDisplay
{
    // 320 x 240
    // Rand 2
    // Kreis: Radius: 60, x: 250, y: 80
    // Rechteck: x: 4, y: 10, w: 126, h: 100
    // Rechteck: x: 4, y: 170, w: 232, h: 110
    bool CleanScreen = true;

public:
    void DrawStartScreen(void)
    {
        CleanScreen = false;
        Display.fillScreen(ILI9341_BLACK);
        Display.drawRect(2, 2, 316, 236, ILI9341_RED); // Rechteck (Display)

        Display.setCursor(4, 4);             // Textposition
        Display.setTextColor(ILI9341_WHITE); // Textcolor
        Display.setTextSize(10);             // Textsize
        Serial.println("Bodensee Devils");   // Text
    }

    void DrawInfoScreen(void)
    {
        CleanScreen = false;
        Display.fillScreen(ILI9341_BLACK);

        Display.drawRect(2, 2, 316, 236, ILI9341_RED);     // Rechteck (Display)
        Display.drawLine(0, 120, 320, 120, ILI9341_GREEN); // horizontale Linie (Display)
        Display.drawLine(160, 0, 160, 240, ILI9341_GREEN); // vertikale Linie (Display)

        Display.setCursor(4, 4);                         // Textposition
        Display.setTextColor(ILI9341_WHITE);             // Textcolor
        Display.setTextSize(2);                          // Textsize
        Display.println("Bodensee Devils - " + version); // Text

        Display.drawCircle(250, 80, 60, ILI9341_BLUE);                    // Kreis (IR/Kompass)
        Display.drawPartialCircle(250, 80, 180, 360, 60, ILI9341_YELLOW); // Kreis (IR area)
        Display.drawPartialCircle(250, 80, 19, 21, 60, ILI9341_RED);      // Kreis (IR Degree)
        Display.drawPartialCircle(250, 80, 359, 1, 60, ILI9341_GREEN);    // Kreis (Goal Degree)
        Display.drawCircle(250, 80, 55, ILI9341_BLACK);                   // Kreis (Schwarze Mitte) evtl obszolet
        Display.drawCircle(250, 50, 4, ILI9341_GREEN);                    // Kreis (LDR)
        Display.setTextSize(1);                                           // Textsize
        Display.setCursor(240, 70);                                       // Textposition
        Display.setTextColor(ILI9341_VIOLET);                             // Textcolor
        Display.println(IR_Heading);                                      // Text (IR_Heading)
        Display.println(OrbitDirection);                                  // Text (OrbitDirection)

        // US
        Display.drawRect(4, 10, 126, 100, ILI9341_YELLOW); // Rechteck (US)
        Display.drawLine(4, 60, 130, 60, ILI9341_YELLOW);  // horizontale Linie (US)
        Display.drawLine(67, 10, 67, 110, ILI9341_YELLOW); // vertikale Linie (US)
        Display.drawCircle(67, 60, 4, ILI9341_DARKBLUE);   // Kreis (US)
        Display.setTextColor(ILI9341_BLUEVIOLET);          // Textcolor
        Display.setTextSize(1);                            // Textsize
        Display.setCursor(69, 12);                         // Textposition (US_Vorne)
        Display.println(US_Vorne);                         // Text (US_Vorne)
        Display.setCursor(69, 88);                         // Textposition (US_Hinten)
        Display.println(US_Hinten);                        // Text (US_Hinten)
        Display.setCursor(108, 62);                        // Textposition (US_Rechts)
        Display.println(US_Rechts);                        // Text (US_Rechts)
        Display.setCursor(6, 62);                          // Textposition (US_Links)
        Display.println(US_Links);                         // Text (US_Links)

        // User input (Switches / Btns)
        Display.drawRect(4, 170, 232, 110, ILI9341_MAGENTA);                    // Rechteck (Switches)
        Display.fillRect(4, 172, 33, 50, SWI1 ? ILI9341_RED : ILI9341_GREEN);   // Rechteck (Switches 1)
        Display.fillRect(37, 172, 33, 50, SWI2 ? ILI9341_RED : ILI9341_GREEN);  // Rechteck (Switches 2)
        Display.fillRect(70, 172, 33, 50, SWI3 ? ILI9341_RED : ILI9341_GREEN);  // Rechteck (Switches 3)
        Display.fillRect(103, 172, 33, 50, SWI4 ? ILI9341_RED : ILI9341_GREEN); // Rechteck (Switches 4)
        Display.fillRect(136, 172, 34, 50, ILI9341_BLACK);                      // Rechteck (PLATZHALTER)
        Display.fillRect(170, 172, 33, 50, SWI5 ? ILI9341_RED : ILI9341_GREEN); // Rechteck (Switches 5)
        Display.fillRect(203, 172, 33, 50, SWI6 ? ILI9341_RED : ILI9341_GREEN); // Rechteck (Switches 6)

        Display.fillRect(4, 228, 33, 50, BTN1 ? ILI9341_RED : ILI9341_GREEN);          // Rechteck (Button 1)
        Display.fillRect(37, 228, 33, 50, BTN2 ? ILI9341_RED : ILI9341_GREEN);         // Rechteck (Button 2)
        Display.fillRect(70, 228, 33, 50, BTN3 ? ILI9341_RED : ILI9341_GREEN);         // Rechteck (Button 3)
        Display.fillRect(103, 228, 33, 50, BTN4 ? ILI9341_RED : ILI9341_GREEN);        // Rechteck (Button 4)
        Display.fillRect(136, 228, 34, 50, ILI9341_BLACK);                             // Rechteck (PLATZHALTER)
        Display.fillRect(170, 228, 33, 50, BTN5 ? ILI9341_RED : ILI9341_GREEN);        // Rechteck (Button 5)
        Display.fillRect(203, 228, 33, 50, MainSwitch ? ILI9341_BLUE : ILI9341_GREEN); // Rechteck (MainSwitch)
    }

    void DrawClearScreen(void)
    {
        if (CleanScreen)
            return;

        Display.fillScreen(ILI9341_BLACK);
        CleanScreen = true;
    }
};

// Zeigt alles relevante auf dem Display an
void Display(void)
{
    if (!DISPLAY_USE)
        return;

    DisplayTime = 0;
    if (SWI1)
        DrawInfoScreen();
    else
        DrawClearScreen();
    DisplayRT = DisplayTime;
}
