public:
    // PID Regler berechnen
    // ausgenommen von rT Regel da unabhängig vom Mainloop
    float calculatePID(compass_sensor_event compass, movement_event movement)
    {
        float PIDKorrektur = compass.orbitDirection - movement.AngleOfAttack; // Abweichung berechnen

        PID_I = PID_I + PIDKorrektur;

        if (PIDKorrektur > -(PID_I_threshhold) && PIDKorrektur < PID_I_threshhold) // wenn I gegen null geht auf 0 setzen
            PID_I = 0;

        if (PID_I > PIDpa_IlimitMax) // I nach oben begrenzen
            PID_I = PIDpa_IlimitMax;
        else if (PID_I < PIDpa_IlimitMin) // I nach unten begrenzen
            PID_I = PIDpa_IlimitMin;

        PID_P = PIDKorrektur * PID_P_Multiplier; // Berechnung P wert

        PID_I = PID_I * PID_I_Multiplier; // Berechnung I wert

        PID_D = (PIDKorrektur - PIDKorrektur_memory) * PID_D_Multiplier; // Berechnung D wert

        float internPID = PID_P + PID_I + PID_D; // PID Wert berechnen
        PIDKorrektur_memory = PIDKorrektur;      // PIDKorrektur speichern für D wert
        return internPID;
    }

    // Setzt alle Motoren
    // @param Motor1: Geschwindigkeit Motor 1
    // @param Motor2: Geschwindigkeit Motor 2
    // @param Motor3: Geschwindigkeit Motor 3
    // @param Dribbleran: Dribbler an/aus
    void setMotors(int Motor1, int Motor2, int Motor3, bool Dribbleran)
    {
        uint16_t speeds[4] =
            {
                (uint16_t)abs(Motor1 * 256 / 100),
                (uint16_t)abs(Motor2 * 256 / 100),
                (uint16_t)abs(Motor3 * 256 / 100),
                (uint16_t)abs(DribbleSpeed * 256 / 100) // Dribbler immer volle Geschwindigkeit
            };

        if (!Dribbleran)
        {
            speeds[3] = 0;
        }

        for (int i = 0; i < 4; i++)
        {
            if (speeds[i] == 0)
            {
                analogWrite(MotorPWMPorts[i], 0);
                digitalWrite(MotorIN1Ports[i], LOW);
                digitalWrite(MotorIN2Ports[i], LOW);
            }

            if (speeds[i] > 0) // Motor rechts rum
            {
                analogWrite(MotorPWMPorts[i], speeds[i]);
                digitalWrite(MotorIN1Ports[i], HIGH);
                digitalWrite(MotorIN2Ports[i], LOW);
            }

            if (speeds[i] < 0) // Motor links rum
            {
                analogWrite(MotorPWMPorts[i], abs(speeds[i]));
                digitalWrite(MotorIN1Ports[i], LOW);
                digitalWrite(MotorIN2Ports[i], HIGH);
            }
        }
    }

    // Alle Motoren(FAHREN) stoppen
    void brake(void)
    {
        setMotors(0, 0, 0, DribblerState);
    }

    // Den Dribbler bedienen

    // @note Dribbler wird NICHT direkt gesetzt -> ERST wenn die Motoren gesetzt werden
    // @note dir wird ignoriert
    void dribbler(bool an, bool dir)
    {
        DribblerState = an;
        // Dribbler_dir = dir; // wird ignoriert -> nicht auskommentieren und eventuell dir invertieren
    }

    // lädt den kicker auf
    void kicker_reset(void)
    {
        if (KickerTimer > KickerExtendTime) // Zeit die der Kicker an ist
        {
            digitalWrite(KICKER_PORT, LOW);
        }
    }

    // Schießt den Ball
    void kick(void)
    {
        if (KickerTimer > KickerExtendTime + KickerCooldown) // Zeit die der Kicker zum Abkühlen hat
        {
            digitalWrite(KICKER_PORT, HIGH);
            KickerTimer = 0;
        }
    }
