    // Distance > 30 Am ROboter
    // Distance < 30 Weiter weg
    // Distance < 35 guter threshhold
    // Evtl Hysterese

    int SchwelleHinten = 36;
    // int SchwelleVorne = 8;

    if (IR_dist_Max < ir.distance)
        IR_dist_Max = ir.distance;

    if (IR_dist_Max < 22) // Ball schwach schwellen deaktivieren
        SchwelleHinten = -1;
    else if (IR_dist_Max < 40)
        SchwelleHinten = 19;
    else
        SchwelleHinten = 36;

    ir.ballanfahrt_neuSF = 1;

    if (ir.Orbit_direction < 0)
        angleToMoveAroundBall = 90;

    if (ir.Orbit_direction > 120 || ir.Orbit_direction < -120) // hinter dem Roboter 120 - 180
    {
        if (ir.distance < SchwelleHinten) // weit weg
            ir.ballanfahrt_neuW = moveDirToBall * 1.3, ir.ballanfahrt_neuSF = 1.2;
        else                                                              // nah dran
            ir.ballanfahrt_neuW = 90 * (ir.Orbit_direction > 0 ? 1 : -1); // moveDirToBall + angleToMoveAroundBall;
    }
    else if (ir.Orbit_direction > 80 || ir.Orbit_direction < -80) // hinten neben dem Roboter 80 - 120
    {
        if (ir.distance < SchwelleHinten) // weit weg
            ir.ballanfahrt_neuW = 180, ir.ballanfahrt_neuSF = 1.2;
        else // nah dran
            ir.ballanfahrt_neuW = moveDirToBall + angleToMoveAroundBall;
    }
    else if (ir.Orbit_direction > 30 || ir.Orbit_direction < -30) // vorne neben dem Roboter (80 - 30 Grad)
    {
        if (ir.distance < SchwelleHinten * 0.8) // weit weg
            ir.ballanfahrt_neuW = ir.Orbit_direction * 1.45;
        else
            ir.ballanfahrt_neuW = 122 * (ir.Orbit_direction < 0 ? 1 : -1); // moveDirToBall + angleToMoveAroundBall;

        ir.ballanfahrt_neuSF = 1.2;
    }
    else if (ir.Orbit_direction > 12 || ir.Orbit_direction < -12) // 30- 12 Grad
        ir.ballanfahrt_neuW = moveDirToBall * 1.45;
    else
        ir.ballanfahrt_neuW = moveDirToBall * 1.1; // 10 - 0 Grad
